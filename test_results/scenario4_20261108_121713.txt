import tkinter as tk

class CalculatorApp(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title('GUI Calculator')
        self.geometry('320x420')
        self.expression = ''
        self.display = tk.Entry(self, font=('Segoe UI', 20), justify='right')
        self.display.pack(fill='x', padx=10, pady=10)
        keys = '789/456*123-0.=+'
        for index, key in enumerate(keys):
            if index % 4 == 0:
                row = tk.Frame(self)
                row.pack(fill='x', padx=6, pady=3)
            action = self.on_equals if key == '=' else (lambda k=key: self.on_press(k))
            tk.Button(row, text=key, font=('Segoe UI', 14), command=action).pack(side='left', fill='both', expand=True, padx=2)
        tk.Button(self, text='C', font=('Segoe UI', 14), command=self.on_clear).pack(fill='x', padx=8, pady=6)

    def on_press(self, key):
        self.expression += key
        self.display.delete(0, tk.END)
        self.display.insert(0, self.expression)

    def on_equals(self):
        try:
            self.expression = str(eval(self.expression))
        except Exception:
            self.expression = 'Error'
        self.display.delete(0, tk.END)
        self.display.insert(0, self.expression)

    def on_clear(self):
        self.expression = ''
        self.display.delete(0, tk.END)

if __name__ == '__main__':
    CalculatorApp().mainloop()
