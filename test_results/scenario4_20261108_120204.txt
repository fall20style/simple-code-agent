import tkinter as tk

class CalculatorApp(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("GUI Calculator")
        self.geometry("320x420")
        self.expression = ""
        self.display = tk.Entry(self, font=("Segoe UI", 20), justify="right")
        self.display.pack(fill="x", padx=10, pady=10)
        # Build a keypad using tk.Button widgets inside tk.Frame rows.
        # Row 1: 7 8 9 /
        # Row 2: 4 5 6 *
        # Row 3: 1 2 3 -
        # Row 4: 0 . = +
        # Row 5: C (full width)
        # Every digit and operator button must call self.on_press(text).
        # The "=" button calls self.on_equals and "C" calls self.on_clear.
        # Make the buttons fill their cells and use a readable font.

    def on_press(self, char):
        self.expression += char
        self.display.delete(0, tk.END)
        self.display.insert(0, self.expression)

    def on_equals(self):
        try:
            result = eval(self.expression)
            self.expression = str(result)
        except Exception:
            self.expression = "Error"
        self.display.delete(0, tk.END)
        self.display.insert(0, self.expression)

    def on_clear(self):
        self.expression = ""
        self.display.delete(0, tk.END)

if __name__ == "__main__":
    CalculatorApp().mainloop()
