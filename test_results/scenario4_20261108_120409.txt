import tkinter as tk

class CalculatorApp(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("GUI Calculator")
        self.geometry("320x420")
        self.expression = ""
        self.display = tk.Entry(self, font=("Segoe UI", 20), justify="right")
        self.display.pack(fill="x", padx=10, pady=10)
        rows = [["7", "8", "9", "/"], ["4", "5", "6", "*"], ["1", "2", "3", "-"], ["0", ".", "=", "+"]]
        for row in rows:
            frame = tk.Frame(self)
            frame.pack(fill="x", padx=6, pady=3)
            for label in row:
                if label == "=":
                    command = self.on_equals
                else:
                    command = (lambda c=label: self.on_press(c))
                tk.Button(frame, text=label, font=("Segoe UI", 14), command=command).pack(side="left", fill="both", expand=True, padx=2)
        tk.Button(self, text="C", font=("Segoe UI", 14), command=self.on_clear).pack(fill="x", padx=8, pady=6)

    def on_press(self, char):
        self.expression += char
        self.display.delete(0, tk.END)
        self.display.insert(0, self.expression)

    def on_equals(self):
        try:
            result = eval(self.expression)
            self.expression = str(result)
        except Exception:
            self.expression = "Error"
        self.display.delete(0, tk.END)
        self.display.insert(0, self.expression)

    def on_clear(self):
        self.expression = ""
        self.display.delete(0, tk.END)

if __name__ == "__main__":
    CalculatorApp().mainloop()
