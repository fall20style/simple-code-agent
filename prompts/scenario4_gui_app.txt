import tkinter as tk

class CounterApp(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title('GUI Counter')
        self.count = 0
        self.label = tk.Label(self, text='0', font=('Segoe UI', 32))
        self.label.pack(padx=40, pady=20)
        tk.Button(self, text='+1', font=('Segoe UI', 16), command=self.increment).pack(pady=10)

    def increment(self):
        self.count += 1
        self.label.config(text=str(self.count))

if __name__ == '__main__':
    CounterApp().mainloop()
